function formatDateKey(d) {
return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
}
function formatTime(value) {
if (!value) return ''
const d = new Date(value)
return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
}
function scoreLevel(score) {
if (score >= 80) return { text: '安全', cls: 'good' }
if (score >= 60) return { text: '注意', cls: 'mid' }
return { text: '高风险', cls: 'bad' }
}

computeSafetyScore(history) {
const scores = history.map(r => r.score ?? r.fullResult?.health_score ?? 0).filter(Boolean)
if (!scores.length) return 0
return Math.round(scores.slice(0, 10).reduce((s, v) => s + v, 0) / Math.min(scores.length, 10))

getSafetyView(score) {
if (score >= 80) return { text: '状态良好', cls: 'good' }
if (score >= 60) return { text: '需要关注', cls: 'mid' }
if (score > 0) return { text: '风险偏高', cls: 'bad' }
return { text: '等待记录', cls: 'empty' }
},
computeDelta(history) {
const now = new Date()
const weekStart = new Date(now)
weekStart.setDate(now.getDate() - 6)
const prevStart = new Date(now)
prevStart.setDate(now.getDate() - 13)
const current = []
const prev = []
history.forEach(r => {
if (!r.createTime) return
const d = new Date(r.createTime)
const score = r.score ?? r.fullResult?.health_score ?? 0
if (!score) return
if (d >= weekStart && d <= now) current.push(score)
else if (d >= prevStart && d < weekStart) prev.push(score)
})
if (!current.length || !prev.length) return '继续积累记录'
const avg = arr => Math.round(arr.reduce((s, v) => s + v, 0) / arr.length)
const diff = avg(current) - avg(prev)
if (diff === 0) return '与上周持平'
return diff > 0 ? `较上周提升 ${diff} 分` : `较上周下降 ${Math.abs(diff)} 分`
},
computeGuardDays(history) {
const days = new Set()
history.forEach(r => {
if (r.createTime) days.add(formatDateKey(new Date(r.createTime)))
})
return days.size
},
computeTodayScanCount(history) {
const todayKey = formatDateKey(new Date())
return history.filter(r => r.createTime && formatDateKey(new Date(r.createTime)) === todayKey).length
},

buildTrend(history) {

.map(r => r.score ?? r.fullResult?.health_score ?? 0).filter(Boolean)

const label = labels[day === 0 ? 6 : day - 1]
result.push({
label,
score,
height: score ? Math.max(28, Math.round(score * 1.15)) : 22,
cls: score >= 80 ? 'good' : score >= 60 ? 'mid' : score > 0 ? 'bad' : 'empty',
})
}
return result

if (!top) return null
return {

def _parse_ingredients(text):
# 配料表标记列表（按优先级）
MARKERS = ["配料表", "配料", "成分表", "成分", "原料表", "原料"]
# 截断标记——遇到这些就停止（配料表之后的内容）
STOP_KW = ["生产日期", "保质期", "储存条件", "贮藏", "贮存", "食用方法",
"产品标准", "生产许可证", "产地", "委托方", "受委托方",
"制造商", "厂家", "服务热线", "客服", "二维码", "扫码",
"微信公众号", "微博", "网址", "www.", "http"]
start = 0
for marker in MARKERS:
idx = text.find(marker)
if idx != -1:
start = idx + len(marker)
break
text = text[start:]
for stop in STOP_KW:
idx = text.find(stop)
if idx != -1:
text = text[:idx]
break
text = re.sub(r"^[：:\s、，,；;\n\r]*", "", text)
parts = re.split(r"[、，,；;。\n\r]+", text)
parts = [p.strip() for p in parts if p.strip()]
FILTER_KW = ['生产', '厂家', '地址', '保质期', '净含量', '营养', '每100',
'执行', '标准', '温度', '储存', '食用方法', '产地',
'服务热线', '热线', '商标', '公司', '有限', '科技',
'生物', '授权', '使用', '许可', '证', '号', 'QS', 'SC',
'℃', '焦', '瓦', '毫克', '克/', 'g/', 'mg/', '%', '日期',
'年', '月', '提示', '请', '避免', '直射', '干燥', '阴凉',
'食用前', '建议', '警告', '注意', '联系', '客服',
'省', '市', '县', '区', '路', '街', '号楼', '传真', '邮编',
'网址', 'www', 'http', '扫码', '红包', '抽奖', '赢', '活动',
'兑换', '积分', '会员', '关注', '分享', '点赞', '投票',
'生产日期', '储存条件', '联系电话', '电话号码', '邮政编码',
'执行标准', '生产许可证', '备案号', '注册号', 'GB', 'SC', 'QS',
'电话', '邮编']
result = []
for p in parts:
# 过长文本通常是地址、说明或宣传语，不作为单个配料处理
if len(p) > 15:
continue
# 命中生产、标准、联系方式等关键词时，判定为非配料字段
if any(kw in p for kw in FILTER_KW):
continue
# 纯数字或纯英文一般不是中文配料名称
if re.match(r'^\d+$', p) or re.match(r'^[a-zA-Z]+$', p):
continue
if re.match(r'^[\d.]+%?$', p) or re.match(r'^[\d.]+(g|mg|ml|kcal|kJ|%)$', p):
continue
if re.search(r'[a-zA-Z]', p) and re.search(r'\d', p) and len(p) > 8:
continue
if re.search(r'\d{3,}', p):
continue
result.append(p)
if len(result) < 2 and parts:
# 至少去掉明显非配料的条目
loose = [p for p in parts if len(p) <= 15
and not re.match(r'^\d+$', p)
and not any(kw in p for kw in FILTER_KW)]
return loose[:20] if loose else parts[:20]
return result
def _get_subcategory(main_category, ingredients_list):
ing_text = ",".join(ingredients_list)
# 饮料类根据碳酸、咖啡因、茶多酚、果汁等关键词细分
if main_category == "饮料类":
if "碳酸" in ing_text or "二氧化碳" in ing_text:
return "碳酸饮料"
if "咖啡因" in ing_text or "牛磺酸" in ing_text:
return "能量饮料"
if "茶" in ing_text or "茶多酚" in ing_text:

def _validate_ingredients(ingredients_list, ocr_text):
if len(ingredients_list) > 80:
return False, "配料表异常（条目过多），请重新拍摄配料表区域"
# 有效配料必须包含食品相关字符
if ingredients_list:
_FOOD_CHARS = ["糖", "粉", "油", "酸", "钠", "胶", "脂", "水", "盐",
"肉", "奶", "乳", "蛋", "豆", "米", "麦",
"淀", "素", "色", "香", "精", "料", "醋", "酱", "芡"]
# 统计食品相关字出现比例，用于判断识别结果是否像真实配料
food_count = sum(1 for ing in ingredients_list if any(c in ing for c in _FOOD_CHARS))
food_ratio = food_count / len(ingredients_list)
if food_ratio < 0.3:
return False, "未找到有效配料表，请对准配料表拍摄"
# 至少1条有效配料即通过
if len(ingredients_list) >= 1:
return True, ""
# 没有识别到任何配料
return False, "无法识别有效配料表，请重新拍摄"
def _run_analysis_pipeline(image_base64, user_type, avoid_ids=None):
from ocr import ocr_image
from classifier import classify
from score import score_ingredients
from ai_analyze import generate_report
# 1. OCR
ocr_text = ocr_image(image_base64=image_base64)
if not ocr_text.strip():
raise ValueError("OCR未能识别到文字")
# 1.5 强制检查原文是否有配料表标识（在解析之前）
_INGREDIENT_PATTERNS = [
r'配\s*料',                          # 配料：/配料表（标点可选）
r'成\s*分[：:;.]',                   # 成分：/成分:（排除"营养成分表"）
r'原\s*料[：:;.]',                   # 原料：/原料:（排除营销用语）
r'配\s*方[：:;.]',                   # 配方：/配方:
r'[Ii]ngredients?',                  # Ingredients/ingredient
]
# 排除——OCR 识别到"营养成分表"但没识别到"配料/原料/配方"
if re.search(r'营\s*养\s*成\s*分', ocr_text) and not re.search(r'配\s*料|原\s*料|配\s*方', ocr_text):
raise ValueError("识别到营养成分表，请翻转包装找到配料表重新拍摄")
# 未命中配料相关标识时，直接中断分析，避免后续评分建立在错误文本上
if not any(re.search(p, ocr_text) for p in _INGREDIENT_PATTERNS):
raise ValueError("未找到配料表，请翻转包装找到以'配料：'开头的文字重新拍摄")
# 2. 解析配料
ingredients_list = _parse_ingredients(ocr_text)
# 2.5 配料可信度验证
valid, reason = _validate_ingredients(ingredients_list, ocr_text)
if not valid:
raise ValueError(reason)
# 3. 分类
print(f"[Classify] 输入配料: {','.join(ingredients_list[:10])}... 共{len(ingredients_list)}条")
food_category_main, confidence = classify(",".join(ingredients_list))
# 分类置信度过低且配料数量不足时，认为图片质量或识别区域不可靠
if confidence < 0.3 and len(ingredients_list) < 2:
raise ValueError("未能识别到有效配料表，请对准配料表重新拍摄")
food_category = _get_subcategory(food_category_main, ingredients_list)
# 4. 评分
score_result = score_ingredients(
ingredients_list,
group_type=user_type,
avoid_ids=avoid_ids,
)
# 5. AI分析
ai_report = generate_report(
food_category=food_category,
health_score=score_result["health_score"],
risk_additives=score_result["risk_additives"],
user_type=user_type,
warnings=score_result["group_warnings"],
)
return {
"ingredients": ",".join(ingredients_list),
"food_category": food_category,
"food_category_main": food_category_main,
"health_score": score_result["health_score"],
"risk_count": len(score_result["risk_additives"]),
"risk_additives": score_result["risk_additives"],
"group_warnings": score_result["group_warnings"],
"beneficial_ingredients": score_result["beneficial_ingredients"],
"beneficial_bonus": score_result.get("beneficial_bonus", 0),
"avoid_warnings": score_result["avoid_warnings"],
"stacking_penalty": score_result.get("stacking_penalty", 0),
"ingredients_count_penalty": score_result.get("ingredients_count_penalty", 0),
"ai_report": ai_report,
}

return "\n".join(words)
except Exception as e:
if attempt < retries:
# 短暂等待后重试，缓解网络抖动或临时接口异常
time.sleep(1)
continue

def _calculate_score(matched_additives, matched_warnings, avoid_warnings, ingredients_count=0, group_type=None, beneficial_ingredients=None):
score = 100  # 基准分
# 排除已知有益添加剂——它们不应被当作风险成分处理
beneficial_names = set(BENEFICIAL_ADDITIVE_NAMES)
if beneficial_ingredients:
beneficial_names |= {b["nutrient"] for b in beneficial_ingredients}
filtered_count = len(matched_additives)
matched_additives = [a for a in matched_additives if a.name not in beneficial_names]
filtered_count -= len(matched_additives)
if filtered_count:
print(f"[Score] 排除有益添加剂 {filtered_count}个", flush=True)
high = [a for a in matched_additives if a.risk_level == "高"]
mid = [a for a in matched_additives if a.risk_level == "中"]
low = [a for a in matched_additives if a.risk_level == "低"]
risk_additives = []
for a in high:
risk_additives.append({"name": a.name, "risk_level": "高", "reason": a.reason, "deduct_score": 20})
for a in mid:
risk_additives.append({"name": a.name, "risk_level": "中", "reason": a.reason, "deduct_score": 10})
for a in low:
risk_additives.append({"name": a.name, "risk_level": "低", "reason": a.reason, "deduct_score": 5})
score -= len(high) * RISK_PENALTY["高"]
score -= len(mid) * RISK_PENALTY["中"]
score -= len(low) * RISK_PENALTY["低"]
score -= len(avoid_warnings) * AVOID_PENALTY
# 配料数量惩罚（累加）
ingredients_count_penalty = 0
if ingredients_count > 15:
# 配料较多时提示加工复杂度偏高
score -= 5
ingredients_count_penalty = 5
if ingredients_count > 25:
# 配料过多时继续累加扣分
score -= 5
ingredients_count_penalty += 5
# 叠加惩罚：同类添加剂（type相同）超过指定数量触发
from collections import Counter
type_counter = Counter(a.type for a in matched_additives if a.type)
stacking_penalty = 0
for t, cnt in type_counter.items():
if cnt > STACK_THRESHOLD:
score -= STACK_PENALTY
stacking_penalty += STACK_PENALTY
# ── 5. 人群额外惩罚 ──
group_extra_penalty = 0
group_extra_details = []
if group_type and group_type in GROUP_PENALTY_RULES:
for additive in matched_additives:
base = RISK_PENALTY.get(additive.risk_level, 0)
if base == 0:
continue
for match_type, match_val, multiplier in GROUP_PENALTY_RULES[group_type]:
matched = False
# 按类型、名称精确匹配或名称包含三种方式触发人群额外惩罚
if match_type == "type" and additive.type == match_val:
matched = True
elif match_type == "name" and additive.name == match_val:
matched = True
elif match_type == "name_contains" and match_val in additive.name:
matched = True
if matched:
extra = int(base * (multiplier - 1))
if extra > 0:
group_extra_penalty += extra
group_extra_details.append({
"name": additive.name,
"risk_level": additive.risk_level,
"base_deduct": base,
"multiplier": multiplier,
"extra_deduct": extra,
})
break  # 每条添加剂只触发一条规则
extra_map = {d["name"]: {"multiplier": d["multiplier"], "extra_deduct": d["extra_deduct"]}
for d in group_extra_details}
for ra in risk_additives:
if ra["name"] in extra_map:
ra["group_multiplier"] = extra_map[ra["name"]]["multiplier"]
ra["group_extra_deduct"] = extra_map[ra["name"]]["extra_deduct"]
score -= group_extra_penalty
score = max(0, min(100, score))
print(f"[Score] 高风险:{len(high)}个x20 中风险:{len(mid)}个x10 低风险:{len(low)}个x5", flush=True)
print(f"[Score] 排除有益添加剂:{len(beneficial_names & orig_count)}个" if hasattr(score, 'orig_count') else "", flush=True)
print(f"[Score] 避免成分:{len(avoid_warnings)}个x5 配料数:{ingredients_count} 叠加惩罚:{stacking_penalty}", flush=True)
print(f"[Score] 人群额外惩罚:{group_extra_penalty}", flush=True)
print(f"[Score] 最终分数:{score}", flush=True)
group_warnings = [{"type": w.warning_type, "msg": w.warning_msg, "suggestion": w.suggestion}
for w in matched_warnings]
return score, {
"risk_additives": risk_additives,
"group_warnings": group_warnings,
"stacking_penalty": stacking_penalty,
"ingredients_count_penalty": ingredients_count_penalty,
"group_extra_penalty": group_extra_penalty,
"group_extra_details": group_extra_details,
}

computeComparison() {

const benefitB = foodB.beneficial_ingredients || []

function getRiskView(level) {
const num = Number(level)
if (num === 1) return { text: '低风险', tone: 'low', width: '33%' }
if (num === 2) return { text: '中风险', tone: 'mid', width: '66%' }
if (num === 3) return { text: '高风险', tone: 'high', width: '100%' }
return { text: '待确认', tone: 'unknown', width: '18%' }
}
function normalizeEntry(entry, fromApi) {
const risk = getRiskView(entry.risk_level)
const alias = (entry.alias || []).filter(item => item && item !== entry.name)
return {
...entry,
fromApi,
riskText: risk.text,
riskTone: risk.tone,
riskWidth: risk.width,
aliasText: alias.join(' / '),
specialGroupsText: (entry.special_groups || []).join('、'),
infoRows: [
{ label: '主要用途', value: entry.usage || '暂无数据' },
{ label: '常见食品', value: entry.common_in || '暂无数据' },
{ label: 'ADI 值', value: entry.adi || '暂无数据' },
{ label: '类别', value: entry.category || '未分类' },
],
}
}

const FAMILY_KEY = 'familyMembers'
const ACTIVE_KEY = 'activeMemberId'
const HISTORY_KEY = 'scanHistory'
function generateId() {
return 'm_' + Date.now() + '_' + Math.random().toString(36).slice(2, 7)
}
function getMembers() {
return wx.getStorageSync(FAMILY_KEY) || []
}
function getActiveMemberId() {
return wx.getStorageSync(ACTIVE_KEY) || ''
}
function getActiveMember() {
const members = getMembers()
const activeId = getActiveMemberId()
return members.find(m => m.memberId === activeId) || members[0] || null
}
function setActiveMember(id) {
wx.setStorageSync(ACTIVE_KEY, id)
}
function addMember(info) {
const members = getMembers()
if (members.length >= 6) {
return { ok: false, msg: '最多添加6位家庭成员' }
}
const member = {
memberId: generateId(),
nickname: info.nickname || '',
gender: info.gender || '',
age: info.age || '',
height: info.height || '',
weight: info.weight || '',
relationship: info.relationship || '',
userType: info.userType || '普通人群',
selectedAvoids: info.selectedAvoids || [],
avatar: info.avatar || '',
createTime: Date.now()
}
members.push(member)
// 家庭成员资料保存在本地缓存，供首页、报告页和扫描页共享
wx.setStorageSync(FAMILY_KEY, members)
return { ok: true, member }
}
function updateMember(id, info) {
const members = getMembers()
const idx = members.findIndex(m => m.memberId === id)
if (idx === -1) return false
if (info.nickname !== undefined) members[idx].nickname = info.nickname
if (info.gender !== undefined) members[idx].gender = info.gender
if (info.age !== undefined) members[idx].age = info.age
if (info.height !== undefined) members[idx].height = info.height
if (info.weight !== undefined) members[idx].weight = info.weight
if (info.relationship !== undefined) members[idx].relationship = info.relationship
if (info.userType !== undefined) members[idx].userType = info.userType
if (info.selectedAvoids !== undefined) members[idx].selectedAvoids = info.selectedAvoids
if (info.avatar !== undefined) members[idx].avatar = info.avatar
wx.setStorageSync(FAMILY_KEY, members)
return true
}
function deleteMember(id) {
const members = getMembers()
if (members.length <= 1) {
return { ok: false, msg: '至少保留一位家庭成员' }
}
const filtered = members.filter(m => m.memberId !== id)
if (filtered.length === members.length) {
return { ok: false, msg: '成员不存在' }
}
wx.setStorageSync(FAMILY_KEY, filtered)
if (getActiveMemberId() === id) {
// 删除当前成员后自动切换到剩余列表第一位
setActiveMember(filtered[0].memberId)
}
return { ok: true }
}
function getFilteredHistory() {
const history = wx.getStorageSync(HISTORY_KEY) || []
const activeId = getActiveMemberId()
if (!activeId) return history
return history.filter(r => !r.memberId || r.memberId === activeId)
}
function addScanRecord(record) {
const activeId = getActiveMemberId()
const fullRecord = Object.assign({}, record, { memberId: activeId })
const history = wx.getStorageSync(HISTORY_KEY) || []
history.unshift(fullRecord)
// 只保留最近50条记录，控制本地缓存体积
if (history.length > 50) history.length = 50
wx.setStorageSync(HISTORY_KEY, history)
return fullRecord
}
function initFamily() {
const members = getMembers()
if (members.length > 0) {
if (!getActiveMemberId()) {
// 已有成员但未设置当前成员时，默认选中第一位
setActiveMember(members[0].memberId)
}
return false
}
// 首次启用家庭功能时，以旧版用户画像创建“我自己”成员
const profile = wx.getStorageSync('userProfile') || {}
const memberId = generateId()
const firstMember = {
memberId: memberId,
nickname: profile.nickname || '我自己',
gender: profile.gender || '',
age: profile.age || '',
height: profile.height || '',
weight: profile.weight || '',
relationship: '我自己',
userType: profile.userType || '普通人群',
selectedAvoids: profile.selectedAvoids || [],
avatar: profile.avatar || '',
createTime: Date.now()
}
wx.setStorageSync(FAMILY_KEY, [firstMember])
wx.setStorageSync(ACTIVE_KEY, memberId)
const history = wx.getStorageSync(HISTORY_KEY) || []
if (history.length > 0) {
// 旧历史记录没有memberId，迁移时统一归属到默认成员
history.forEach(function (r) {
if (!r.memberId) r.memberId = memberId
})
wx.setStorageSync(HISTORY_KEY, history)
}
return true
}
module.exports = {
getMembers: getMembers,
getActiveMemberId: getActiveMemberId,
getActiveMember: getActiveMember,
setActiveMember: setActiveMember,
addMember: addMember,
updateMember: updateMember,
deleteMember: deleteMember,
getFilteredHistory: getFilteredHistory,
addScanRecord: addScanRecord,
initFamily: initFamily
}
const app = getApp()
const BASE_URL = app.globalData.baseUrl
function _request(url, method, data) {
return new Promise((resolve, reject) => {
wx.request({
url: BASE_URL + url,
method: method,
data: data,
timeout: 30000,
header: { 'Content-Type': 'application/json' },
success(res) {
if (res.data.code === 200) {
// 后端统一返回code=200时取data字段作为业务结果
resolve(res.data.data)
} else {
reject({ msg: res.data.msg || '请求失败', code: res.data.code })
}
},
fail(err) {
reject({ msg: '网络错误: ' + err.errMsg, code: -1 })
},
})
})
}
function analyze(image, userType, avoidIds) {
return _request('/analyze', 'POST', {
image,
user_type: userType || app.globalData.userType,
avoid_ids: avoidIds ? avoidIds.join(',') : undefined,
user_id: app.globalData.userId,
})
}
function compare(image1, image2, userType) {
return _request('/compare', 'POST', {
image1,
image2,
user_type: userType || app.globalData.userType,
})
}
function getHistory() {
return _request('/history?user_id=' + app.globalData.userId, 'GET')
}
function getReport(type, date) {
if (type === 'daily') {
// 日报使用专用接口，包含AI日总结等字段
return _request('/daily?user_id=' + app.globalData.userId, 'GET')
}
return _request('/history?user_id=' + app.globalData.userId + '&type=' + type + '&date=' + date, 'GET')
}
function compareReport(type, date1, date2) {
return Promise.all([
getReport(type, date1),
getReport(type, date2),
]).then(([r1, r2]) => ({ report1: r1, report2: r2 }))
}
function aiChat(question, history) {
return _request('/api/ai-chat', 'POST', {
question,
history: history || [],
user_type: app.globalData.userType,
user_id: app.globalData.userId,
})
}
function saveProfile(profile) {
wx.setStorageSync('userProfile', profile)
const g = app.globalData
g.userId = profile.userId
g.userType = profile.userType
g.hasProfile = true
return Promise.resolve({ success: true })
}
module.exports = {
analyze,
compare,
getHistory,
getReport,
compareReport,
aiChat,
saveProfile,
}
